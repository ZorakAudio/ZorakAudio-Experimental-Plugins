cmake_minimum_required(VERSION 3.24)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(JSFXNativeFrontend VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ZA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_subdirectory("${ZA_ROOT}/libs/JUCE" juce EXCLUDE_FROM_ALL)
set(WDL_SOURCES)
foreach(name nseel-compiler nseel-eval nseel-ram nseel-cfunc nseel-lextab nseel-caltab nseel-yylex)
  list(APPEND WDL_SOURCES "${ZA_ROOT}/src/WDL/eel2/${name}.c")
endforeach()
if(WIN32)
  list(APPEND WDL_SOURCES "${ZA_ROOT}/src/WDL/win32_utf8.c")
endif()
add_library(jsfx_native_frontend STATIC Frontend.cpp SourceResolver.cpp ${WDL_SOURCES})
target_include_directories(jsfx_native_frontend PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${ZA_ROOT}/src")
target_compile_definitions(jsfx_native_frontend PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 PRIVATE EEL_TARGET_PORTABLE=1 WDL_FFT_REALSIZE=8)
target_link_libraries(jsfx_native_frontend PUBLIC juce::juce_core)
add_executable(jsfx_frontend Main.cpp)
target_link_libraries(jsfx_frontend PRIVATE jsfx_native_frontend)
add_executable(jsfx_frontend_check NativeCheck.cpp)
target_link_libraries(jsfx_frontend_check PRIVATE jsfx_native_frontend)
if(WIN32)
  if(MSVC)
    target_link_options(jsfx_frontend PRIVATE "/STACK:8388608")
    target_link_options(jsfx_frontend_check PRIVATE "/STACK:8388608")
  else()
    target_link_options(jsfx_frontend PRIVATE "-Wl,/STACK:8388608")
    target_link_options(jsfx_frontend_check PRIVATE "-Wl,/STACK:8388608")
  endif()
endif()
enable_testing()
add_test(NAME native_frontend_contract COMMAND jsfx_frontend_check)
