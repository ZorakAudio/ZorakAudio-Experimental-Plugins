cmake_minimum_required(VERSION 3.22)
project(ImportExtractorChecks LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ZA_ROOT "../.." ABSOLUTE)
add_subdirectory("${ZA_ROOT}/libs/JUCE" juce)
add_executable(extractor_detector_check detector_test.cpp)
target_include_directories(extractor_detector_check PRIVATE "${ZA_ROOT}/src")
foreach(kind editor material combined)
  add_executable(extractor_${kind}_check ${kind}_test.cpp)
  target_include_directories(extractor_${kind}_check PRIVATE "${ZA_ROOT}/src")
  target_compile_definitions(extractor_${kind}_check PRIVATE JUCE_MODAL_LOOPS_PERMITTED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
  target_link_libraries(extractor_${kind}_check PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags)
endforeach()
enable_testing()
add_test(NAME detector COMMAND extractor_detector_check "${CMAKE_CURRENT_BINARY_DIR}/detector-results.csv")
add_test(NAME editor COMMAND extractor_editor_check "${CMAKE_CURRENT_BINARY_DIR}/editor-results")

add_test(NAME combined COMMAND extractor_combined_check "${CMAKE_CURRENT_BINARY_DIR}/combined-results")

# Optional real-recording analysis/capture: pass input and output directories.
add_executable(extractor_recording_probe recording_probe.cpp)
target_include_directories(extractor_recording_probe PRIVATE "${ZA_ROOT}/src")
target_compile_definitions(extractor_recording_probe PRIVATE JUCE_MODAL_LOOPS_PERMITTED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(extractor_recording_probe PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags)
