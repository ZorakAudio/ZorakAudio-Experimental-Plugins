cmake_minimum_required(VERSION 3.22)
project(JsfxShowcase LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(REPO "${CMAKE_CURRENT_LIST_DIR}/../..")
set(WDL "${REPO}/src/WDL")
find_package(Python3 COMPONENTS Interpreter REQUIRED)
option(ZA_SHOWCASE_ASAN "Instrument headless tests and WDL/LICE with AddressSanitizer" OFF)
if(ZA_SHOWCASE_ASAN)
 if(MSVC)
  message(FATAL_ERROR "Headless ASan runner currently requires Clang/GCC; production MSVC build is unaffected")
 endif()
 add_compile_options(-fsanitize=address -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address -fno-omit-frame-pointer)
endif()
if(NOT SHOWCASE_GENERATED)
  message(FATAL_ERROR "Set SHOWCASE_GENERATED to the generated DSP headers/object directory")
endif()
add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/numeric_runtime.inc"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/extract_numeric_runtime.py"
    "${REPO}/src/JSFXJuceProcessor.cpp" "${CMAKE_BINARY_DIR}/numeric_runtime.inc"
  DEPENDS "${REPO}/src/JSFXJuceProcessor.cpp" extract_numeric_runtime.py VERBATIM)
add_custom_target(showcase_numeric_runtime DEPENDS "${CMAKE_BINARY_DIR}/numeric_runtime.inc")
add_library(showcase_eel STATIC
  "${WDL}/eel2/nseel-compiler.c" "${WDL}/eel2/nseel-eval.c"
  "${WDL}/eel2/nseel-yylex.c" "${WDL}/eel2/nseel-ram.c" "${WDL}/eel2/nseel-cfunc.c" "${WDL}/fft.c")
target_compile_definitions(showcase_eel PUBLIC EEL_TARGET_PORTABLE=1 WDL_FFT_REALSIZE=8)
if(MSVC)
 set(OBJ_SUFFIX obj)
else()
 set(OBJ_SUFFIX o)
endif()
add_executable(dsp_reference dsp_reference.cpp
  "${SHOWCASE_GENERATED}/JSFXDSP.${OBJ_SUFFIX}")
add_dependencies(dsp_reference showcase_numeric_runtime)
target_include_directories(dsp_reference PRIVATE "${REPO}/src" "${SHOWCASE_GENERATED}" "${CMAKE_BINARY_DIR}")
target_link_libraries(dsp_reference PRIVATE showcase_eel)
if(NOT MSVC)
 target_link_libraries(dsp_reference PRIVATE pthread m)
endif()
add_executable(eel_eval eel_eval.cpp)
target_include_directories(eel_eval PRIVATE "${REPO}/src")
target_link_libraries(eel_eval PRIVATE showcase_eel)
if(NOT MSVC)
 target_link_libraries(eel_eval PRIVATE pthread m)
endif()
add_executable(gfx_showcase gfx_showcase.cpp
  "${SHOWCASE_GENERATED}/JSFXDSP.${OBJ_SUFFIX}")
add_dependencies(gfx_showcase showcase_numeric_runtime)
target_include_directories(gfx_showcase PRIVATE "${REPO}/src" "${SHOWCASE_GENERATED}" "${CMAKE_BINARY_DIR}")
target_link_libraries(gfx_showcase PRIVATE showcase_eel)
if(NOT MSVC)
 target_link_libraries(gfx_showcase PRIVATE pthread m)
endif()
if(ZA_SHOWCASE_JUCE_ROOT)
 add_subdirectory("${ZA_SHOWCASE_JUCE_ROOT}" "${CMAKE_BINARY_DIR}/JUCE")
 target_compile_definitions(gfx_showcase PRIVATE ZA_SHOWCASE_REAL_JUCE=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
 target_link_libraries(gfx_showcase PRIVATE juce::juce_audio_basics juce::juce_graphics)
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
 # The AOT generator emits a non-PIC object. Avoid text relocations in test PIEs.
 target_link_options(dsp_reference PRIVATE -no-pie)
 target_link_options(gfx_showcase PRIVATE -no-pie)
endif()
